67def build_frame(rw, nn, aa, cc, data):
68 frame_bytes = [rw, nn, aa, cc] + data
69 cs = sum(frame_bytes) & 0xFFFF
70 return "".join(f"{b:02X}" for b in frame_bytes) + f"{cs:04X}"

74def send_cmd(ser, hexstr, timeout=1.0):
75 wire = (hexstr + "\r").encode("ascii")
76 ser.reset_input_buffer()
77 ser.write(wire)
78 deadline = time.time() + timeout
79 buf = b""
80 while time.time() < deadline:
81 b = ser.read(1)
82 if b in (b"\r", b"\n"):
83 if buf:
84 break
85 elif b:
86 buf += b
87 return buf.decode("ascii", errors="replace").strip()

91def parse_response(resp):
92 if len(resp) < 10 or len(resp) % 2 != 0:
93 return False, 0, 0, []
94 try:
95 raw = bytes.fromhex(resp)
96 except ValueError:
97 return False, 0, 0, []
98 rw = raw[0]; nn = raw[1]; cc = raw[3]
99 if len(raw) != 4 + nn + 2:
100 return False, rw, cc, []
101 data = list(raw[4:4+nn])
102 cs_recv = (raw[-2] << 8) | raw[-1]
103 cs_calc = sum(raw[:-2]) & 0xFFFF
104 ok = (cs_recv == cs_calc) and rw in (0x02, 0x03)
105 return ok, rw, cc, data

109def _write(ser, cc, data):

246def read_power_meter(ser):
247 """0x15 Power Monitor - laser returns 4 bytes, units 0.01W.
248 Likely LDD electrical input power, not optical output."""
249 ok, data, _ = _read(ser, 0x15)

251 return None

253 val = (data[0]<<24)|(data[1]<<16)|(data[2]<<8)|data[3]

507def _sim_drift(val, target, rate=0.05):
508 """Drift val toward target with small noise."""
509 val += (target - val) * rate
510 val += random.uniform(-0.02, 0.02)
511 return round(val, 2)

541def sim_write(cc, data):
542 if cc == 0x04: # shutter
543 if data[0] == 0x01:
544 _sim.shutter = True
545 else:
546 _sim.shutter = False
547 return True
548 elif cc == 0x10: # LDD
549 _sim.ldd = bool(data[0])
550 return True
551 elif cc == 0x11: # set current
552 val = ((data[0] << 8) | data[1]) / 100.0
553 if val <= _sim.max_current:
554 _sim.current_set = val
555 return True
556 return False
557 elif cc == 0x14: # max current
558 _sim.max_current = ((data[0] << 8) | data[1]) / 100.0
559 return True
560 elif cc in (0x07, 0x20, 0x23, 0x24, 0x56, 0x69):
561 return True # accept silently
562 elif cc == 0x54: # duty width
563 _sim.duty_width = (data[0] << 16) | (data[1] << 8) | data[2]
564 return True
565 return False

1083 while self._poll_running:
1084 if SIM_MODE:
1085 try:
1086 d = {
1087 'status': sim_read_status(),
1088 'temp_shg': _sim.shg,
1089 'temp_thg': _sim.thg,
1090 'temp_cb': None,
1091 'act_current': round(_sim.current_act + random.uniform(-0.01,0.01), 2),
1092 'set_current': _sim.current_set,
1093 'max_current': MAX_CURRENT,
1094 'diode_voltage': None,
1095 'prf': _sim.prf if _sim.prf else 0,
1096 'prf_source': 1,
1097 'gate_source': 0,
1098 'duty_mode': 0,
1099 'duty_width': _sim.duty_width,
1100 'flow': None,
1101 'power_w': None,
1102 'soft_faults': None,
1103 'board_faults':None,
1104 }
1105 self.root.after(0, lambda d=d: self._update_ui(d))
1106 except Exception:
1107 pass
1108 elif self.ser and self.ser.is_open:
1109 try:
1110 with self._lock:
1111 d = poll_all(self.ser)
1112 self.root.after(0, lambda d=d: self._update_ui(d))
1113 except Exception:
1114 pass
1115 interval = self._poll_interval
1116 if interval == 0:
1117 # Paused - check every 0.2s for unpause without burning CPU
1118 while self._poll_interval == 0 and self._poll_running:
1119 time.sleep(0.2)
1120 else:
1121 time.sleep(interval)
